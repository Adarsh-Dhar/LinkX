🚀 QUICK START
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

START ALL THREE PROVIDERS (in separate terminals):

  Terminal 1:  cd server && node index.js
  Terminal 2:  cd server && PROVIDER_ID=premium node index.js
  Terminal 3:  cd server && PROVIDER_ID=scam node index.js


TEST THE PROVIDERS:

  curl http://localhost:3050/market/price/CRO    # Get price
  curl http://localhost:3050/alpha/insight/CRO   # Get paywall
  curl http://localhost:3050/health               # Check health

  curl http://localhost:3051/alpha/insight/CRO   # Different price!
  curl http://localhost:3052/alpha/insight/CRO   # Different price!

🔧 ADDING YOUR OWN PROVIDER
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Edit providers.json and add:

{
  "my_provider": {
    "name": "My Custom Provider",
    "port": 3053,
    "wallet": "0xMyWallet",
    "price": "0.5",
    "bias": "bullish"
  }
}

Then start it:

  PROVIDER_ID=my_provider node index.js
